    <h2 class="section-title fade-up">What Makes a Marketing <em>Case Study Worth Trusting</em></h2>
    <div class="rule fade-up"></div>
    <div class="fade-up"><p><a href="https://www.adobe.com/express/learn/blog/marketing-case-study" target="_blank" rel="noopener">Adobe</a> defines a marketing case study as &ldquo;a detailed examination and analysis of a specific strategy, initiative, or marketing campaign.&rdquo; <a href="https://www.canva.com/resources/marketing-case-study/" target="_blank" rel="noopener">Canva</a> describes it as a story of how a business &ldquo;encountered a real-world challenge and used strategic solutions&rdquo; to get results.</p>
<p>For a tool that tells you where to put your ad budget, the bar is higher than a good story. You should be able to see how the results were measured, what went wrong along the way, and whether the numbers hold up when someone else checks them. <section id="challenge" style="background:#071530!important;border-top:1px solid #102040;border-bottom:1px solid #102040">
  <div class="section-inner prose">
    <div class="eyebrow fade-up">The Challenge</div>
    <h2 class="section-title fade-up">The Challenge: <em>Trusting a Decision Engine With Real Money</em></h2>
    <div class="rule fade-up"></div>
    <div class="fade-up"><p>CDAI calculates the real profit of every campaign, after ad spend, fees, partner payouts, refunds, chargebacks, compliance costs, and operating costs you add, then makes one clear decision per campaign every night: scale, hold, cut, or pause. If those decisions are wrong, a business moves money in the wrong direction.</p>
<p>So before CDAI touched a real client&rsquo;s budget, it had to prove two things:</p>
<ol class="steps">
  <li><span><strong>The math is right.</strong> Every cost, lead, and sale adds up the same way no matter how you calculate it.</span></li>
  <li><span><strong>The decisions are right.</strong> When CDAI says scale, the campaign actually deserved more money; when it says cut, the campaign really was losing.</span></li>
</ol></div>
  </div>
</section>

    if (el.closest("[class$='-final']")) return "final_cta";
    var h = null, node = el;
    while (node && node !== document.body) {
      var prev = node.previousElementSibling;
      while (prev) { if (/^H2$/.test(prev.tagName) && prev.id) { h = prev.id; break; } prev = prev.previousElementSibling; }
      if (h) break;
      node = node.parentElement;
    }
    return h || "intro";
  }
  function label(el) { return (el.getAttribute("aria-label") || el.textContent || "").replace(/\s+/g, " ").trim().slice(0, 80); }

  // Every link and button click.
  document.addEventListener("click", function (e) {
    var el = e.target.closest && e.target.closest("a, button, summary");
    if (!el) return;
    if (el.tagName === "SUMMARY") { track("faq_open", { question: label(el) }); return; }
    var href = el.getAttribute("href") || "";
    var p = { link_text: label(el), link_url: href, section: sectionOf(el) };
    if (/cdai-portal\.vercel\.app\/signup/.test(href)) track("signup_click", p);
    else if (/cdai-portal\.vercel\.app/.test(href)) track("portal_click", p);
    else if (/\/pricing/.test(href)) track("pricing_click", p);
    else if (/contribution-margin-marketing/.test(href)) track("validation_report_click", p);
    else if (/^#/.test(href)) track("toc_click", p);
    else if (/^https?:\/\//.test(href) && href.indexOf(location.hostname) === -1) track("outbound_click", p);
    else if (href) track("internal_link_click", p);
    else track("button_click", p);
  }, true);

  // How far people read (25/50/75/100%), each once per visit.
  var marks = [25, 50, 75, 100], hit = {};
  function onScroll() {
    var doc = document.documentElement;
    var pct = Math.round(((window.scrollY || doc.scrollTop) + window.innerHeight) / doc.scrollHeight * 100);
    marks.forEach(function (m) { if (pct >= m && !hit[m]) { hit[m] = true; track("scroll_depth", { percent: m }); } });
  }
  window.addEventListener("scroll", onScroll, { passive: true });

  // Which sections people reach (each H2, once per visit).
  if ("IntersectionObserver" in window) {
    var seen = {};
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (en) {
        var id = en.target.id;
        if (en.isIntersecting && id && !seen[id]) { seen[id] = true; track("section_view", { section: id }); io.unobserve(en.target); }
      });
    }, { threshold: 0.6 });
    document.querySelectorAll("h2[id]").forEach(function (h) { io.observe(h); });
  }
})();
